# Standalone Python Reference Script (faq_nlp_bot.py)
import sys, re, math
import numpy as np

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Sample FAQ Knowledge Base
FAQ_DATASET = [
    {"question": "How long does shipping take and how much does it cost?", "answer": "Standard shipping takes 3-5 business days ($4.99)."},
    {"question": "What is your return and refund policy?", "answer": "We offer a 30-day money-back guarantee for unused items."},
    {"question": "How can I track my order status?", "answer": "Check your email for the shipment tracking link."},
    {"question": "What payment methods do you accept?", "answer": "We accept Visa, Mastercard, PayPal, and Apple Pay."},
    {"question": "How do I reset my account password?", "answer": "Click 'Forgot Password' on the login page to receive a reset link."}
]

class PythonFAQBot:
    def __init__(self, faqs=FAQ_DATASET):
        self.faqs = faqs
        self.vectorizer = TfidfVectorizer(stop_words='english')
        self.questions = [f["question"] for f in faqs]
        self.matrix = self.vectorizer.fit_transform(self.questions)

    def match_question(self, user_query):
        query_vec = self.vectorizer.transform([user_query])
        similarities = cosine_similarity(query_vec, self.matrix).flatten()
        best_idx = np.argmax(similarities)
        score = float(similarities[best_idx])
        
        return {
            "question": self.faqs[best_idx]["question"],
            "answer": self.faqs[best_idx]["answer"] if score >= 0.25 else "No matching FAQ found.",
            "score": score
        }

bot = PythonFAQBot()
print(bot.match_question("How do I get a refund for my order?"))
